Experiment 5 — Conditional Statements
Four tasks: even/odd, largest of three, leap year, and withdrawal tax/balance.
#!/bin/bash
echo "Enter a number:"
read num
if [ $((num%2)) -eq 0 ]; then
    echo "$num is Even"
else
    echo "$num is Odd"
fi
echo "Enter three numbers:"
read a b c
if [ "$a" -ge "$b" ] && [ "$a" -ge "$c" ]; then
    echo "$a is the largest"
elif [ "$b" -ge "$a" ] && [ "$b" -ge "$c" ]; then
    echo "$b is the largest"
else
    echo "$c is the largest"
fi
echo "Enter a year:"
read year
if [ $((year%400)) -eq 0 ] || {
   [ $((year%4)) -eq 0 ] && [ $((year%100)) -ne 0 ]; }; then
    echo "$year is a Leap Year"
else
    echo "$year is not a Leap Year"
fi
echo "Enter account balance:"
read balance
echo "Enter withdrawal amount:"
read withdraw
if [ "$balance" -lt "$withdraw" ]; then
    echo "Insufficient balance"
else
    if [ "$withdraw" -lt 1500 ]; then
        tax=$(echo "$withdraw*0.03" | bc)
    elif [ "$withdraw" -ge 1500 ] && [ "$withdraw" -lt 3000 ]; then
        tax=$(echo "$withdraw*0.04" | bc)
    else
        tax=$(echo "$withdraw*0.05" | bc)
    fi
    total=$(echo "$withdraw+$tax" | bc)
    remaining=$(echo "$balance-$total" | bc)
    echo "Amount withdrawn: $withdraw"
    echo "Tax deducted: $tax"
    echo "Remaining balance: $remaining"
fi
Operators: -eq equal, -ne not equal, -gt greater, -lt less, -ge greater/equal, -le less/equal. Leap year: divisible by 400 OR
divisible by 4 but not by 100.

Experiment 7 — Switch Statement — case/esac Unit Converter
Menu-driven converter with temperature, length, weight, history, clear-history and exit options. Uses bc for decimals.
#!/bin/bash
HISTORY_FILE="$HOME/unit_converter_history.txt"
touch "$HISTORY_FILE"
log_history() {
    echo "$(date '+%Y-%m-%d %H:%M:%S') | $1" >> "$HISTORY_FILE"
}
show_history() {
    echo "=== CONVERSION HISTORY ==="
    if [ ! -s "$HISTORY_FILE" ]; then
        echo "No conversions yet."
    else
        cat -n "$HISTORY_FILE"
    fi
}
while true; do
    echo "1.Temperature  2.Length  3.Weight"
    echo "4.Show History  5.Clear History  6.Exit"
    read -p "Enter choice: " choice
    case $choice in
      1)
        echo "1.Celsius to Fahrenheit  2.Fahrenheit to Celsius"
        read -p "Sub-choice: " sub
        case $sub in
          1) read -p "Celsius: " c
             f=$(echo "scale=2; $c*9/5+32" | bc)
             echo "Fahrenheit: $f"
             log_history "Temperature: ${c}C -> ${f}F" ;;
          2) read -p "Fahrenheit: " f
             c=$(echo "scale=2; ($f-32)*5/9" | bc)
             echo "Celsius: $c"
             log_history "Temperature: ${f}F -> ${c}C" ;;
          *) echo "Invalid sub-choice" ;;
        esac ;;
      2)
        echo "1.Kilometers to Miles  2.Miles to Kilometers"
        read -p "Sub-choice: " sub
        case $sub in
          1) read -p "Kilometers: " km
             miles=$(echo "scale=2; $km*0.621371" | bc)
             echo "Miles: $miles"
             log_history "Length: ${km}km -> ${miles}miles" ;;
          2) read -p "Miles: " miles
             km=$(echo "scale=2; $miles*1.60934" | bc)
             echo "Kilometers: $km"
             log_history "Length: ${miles}miles -> ${km}km" ;;
          *) echo "Invalid sub-choice" ;;
        esac ;;
      3)
        echo "1.Kilograms to Pounds  2.Pounds to Kilograms"
        read -p "Sub-choice: " sub
        case $sub in
          1) read -p "Kilograms: " kg
             lbs=$(echo "scale=2; $kg*2.20462" | bc)
             echo "Pounds: $lbs"
             log_history "Weight: ${kg}kg -> ${lbs}lbs" ;;
          2) read -p "Pounds: " lbs
             kg=$(echo "scale=2; $lbs*0.453592" | bc)
             echo "Kilograms: $kg"
             log_history "Weight: ${lbs}lbs -> ${kg}kg" ;;
          *) echo "Invalid sub-choice" ;;
        esac ;;
      4) show_history ;;
      5) > "$HISTORY_FILE"; echo "History cleared." ;;
      6) echo "Exiting. Thank you!"; break ;;
      *) echo "Invalid choice. Please try again." ;;
    esac
    echo
done

Experiment 8 — FCFS CPU Scheduling (C)
First Come First Served. Processes execute in the order entered. Input is burst time only, as in the manual.
#include <stdio.h>
int main(void) {
    int n, i, j, bt[20], wt[20], tat[20];
    float avgwt = 0, avgtat = 0;
    printf("Enter number of processes: ");
    scanf("%d", &n);
    printf("Enter burst time for each process:\n");
    for (i = 0; i < n; i++) {
        printf("P%d: ", i + 1);
        scanf("%d", &bt[i]);
    }
    wt[0] = 0;
    for (i = 1; i < n; i++) {
        wt[i] = 0;
        for (j = 0; j < i; j++)
            wt[i] += bt[j];
    }
    for (i = 0; i < n; i++)
        tat[i] = bt[i] + wt[i];
    printf("\nProcess\tBT\tWT\tTAT\n");
    for (i = 0; i < n; i++) {
        avgwt += wt[i];
        avgtat += tat[i];
        printf("P%d\t%d\t%d\t%d\n", i+1, bt[i], wt[i], tat[i]);
    }
    printf("Average WT = %.2f\n", avgwt/n);
    printf("Average TAT = %.2f\n", avgtat/n);
    return 0;
}
Formulas: WT of first process = 0; WT[i] = sum of previous burst times; TAT = WT + BT. For the manual's BT 24,3,7,13,21:
average WT = 26.40 and average TAT = 40.00.

Experiment 9A — Priority Scheduling — Non-preemptive (C)
Lower priority number means higher priority. Once a process starts, it runs to completion. Tie-break: earlier arrival.
#include <stdio.h>
#define MAX 20
struct Process { int pid, at, bt, pr, ct, tat, wt, done; };
int main(void) {
    struct Process p[MAX], temp;
    int n, i, time = 0, completed = 0, idx;
    float total_tat = 0, total_wt = 0;
    printf("Enter number of processes: ");
    scanf("%d", &n);
    for (i = 0; i < n; i++) {
        p[i].pid = i + 1;
        printf("AT BT Priority of P%d: ", i+1);
        scanf("%d%d%d", &p[i].at, &p[i].bt, &p[i].pr);
        p[i].done = 0;
    }
    while (completed < n) {
        idx = -1;
        for (i = 0; i < n; i++) {
            if (!p[i].done && p[i].at <= time) {
                if (idx == -1 || p[i].pr < p[idx].pr ||
                   (p[i].pr == p[idx].pr && p[i].at < p[idx].at))
                    idx = i;
            }
        }
        if (idx == -1) { time++; continue; }
        time += p[idx].bt;
        p[idx].ct = time;
        p[idx].tat = p[idx].ct - p[idx].at;
        p[idx].wt = p[idx].tat - p[idx].bt;
        p[idx].done = 1;
        completed++;
    }
    printf("\nPID\tAT\tBT\tPR\tCT\tTAT\tWT\n");
    for (i = 0; i < n; i++) {
        printf("P%d\t%d\t%d\t%d\t%d\t%d\t%d\n",
          p[i].pid,p[i].at,p[i].bt,p[i].pr,p[i].ct,p[i].tat,p[i].wt);
        total_tat += p[i].tat; total_wt += p[i].wt;
    }
    printf("Average TAT = %.2f\n", total_tat/n);
    printf("Average WT = %.2f\n", total_wt/n);
    return 0;
}

Experiment 9B — Priority Scheduling — Preemptive (C)
At every time unit, run the arrived process with the smallest priority number and positive remaining time.
#include <stdio.h>
#define MAX 20
struct Process { int pid, at, bt, rt, pr, ct, tat, wt; };
int main(void) {
    struct Process p[MAX];
    int n, i, time = 0, completed = 0, idx;
    float total_tat = 0, total_wt = 0;
    printf("Enter number of processes: ");
    scanf("%d", &n);
    for (i = 0; i < n; i++) {
        p[i].pid = i+1;
        printf("AT BT Priority of P%d: ", i+1);
        scanf("%d%d%d", &p[i].at, &p[i].bt, &p[i].pr);
        p[i].rt = p[i].bt;
    }
    while (completed < n) {
        idx = -1;
        for (i = 0; i < n; i++) {
            if (p[i].at <= time && p[i].rt > 0) {
                if (idx == -1 || p[i].pr < p[idx].pr ||
                   (p[i].pr == p[idx].pr && p[i].at < p[idx].at))
                    idx = i;
            }
        }
        if (idx == -1) { time++; continue; }
        p[idx].rt--;
        time++;
        if (p[idx].rt == 0) {
            p[idx].ct = time;
            p[idx].tat = time - p[idx].at;
            p[idx].wt = p[idx].tat - p[idx].bt;
            total_tat += p[idx].tat;
            total_wt += p[idx].wt;
            completed++;
        }
    }
    printf("\nPID\tAT\tBT\tPR\tCT\tTAT\tWT\n");
    for (i = 0; i < n; i++)
        printf("P%d\t%d\t%d\t%d\t%d\t%d\t%d\n",
          p[i].pid,p[i].at,p[i].bt,p[i].pr,p[i].ct,p[i].tat,p[i].wt);
    printf("Average TAT = %.2f\n", total_tat/n);
    printf("Average WT = %.2f\n", total_wt/n);
    return 0;
}

Experiment 10A — SJF Scheduling — Non-preemptive (C)
Among arrived unfinished processes, select the smallest burst time and run it to completion.
#include <stdio.h>
struct Process { int pid, at, bt, ct, tat, wt, done; };
int main(void) {
    struct Process p[20];
    int n, i, time = 0, completed = 0, idx;
    float totalTAT = 0, totalWT = 0;
    printf("Enter number of processes: ");
    scanf("%d", &n);
    for (i = 0; i < n; i++) {
        p[i].pid = i+1;
        printf("Arrival Time and Burst Time for P%d: ", i+1);
        scanf("%d%d", &p[i].at, &p[i].bt);
        p[i].done = 0;
    }
    while (completed < n) {
        idx = -1;
        for (i = 0; i < n; i++) {
            if (!p[i].done && p[i].at <= time &&
               (idx == -1 || p[i].bt < p[idx].bt))
                idx = i;
        }
        if (idx == -1) { time++; continue; }
        time += p[idx].bt;
        p[idx].ct = time;
        p[idx].tat = p[idx].ct - p[idx].at;
        p[idx].wt = p[idx].tat - p[idx].bt;
        p[idx].done = 1;
        completed++;
        totalTAT += p[idx].tat;
        totalWT += p[idx].wt;
    }
    printf("\nPID\tAT\tBT\tCT\tTAT\tWT\n");
    for (i = 0; i < n; i++)
        printf("P%d\t%d\t%d\t%d\t%d\t%d\n",
          p[i].pid,p[i].at,p[i].bt,p[i].ct,p[i].tat,p[i].wt);
    printf("Average TAT = %.2f\n", totalTAT/n);

Experiment 11 — LRU Page Replacement (C)
Use frames and a last-used timestamp to replace the least recently used page.
#include <stdio.h>
#define MAX 50
int main(void) {
    int frames, n, pages[MAX], frame[MAX], last[MAX];
    int faults = 0, hits = 0, i, j;
    printf("Enter number of frames: ");
    scanf("%d", &frames);
    printf("Enter number of pages: ");
    scanf("%d", &n);
    printf("Enter reference string: ");
    for (i = 0; i < n; i++) scanf("%d", &pages[i]);
    for (i = 0; i < frames; i++) {
        frame[i] = -1;
        last[i] = -1;
    }
    printf("\nPage\tFrames\t\tResult\n");
    for (i = 0; i < n; i++) {
        int found = 0, pos = -1;
        for (j = 0; j < frames; j++) {
            if (frame[j] == pages[i]) {
                found = 1; pos = j; break;
            }
        }
        if (found) {
            hits++;
            last[pos] = i;
        } else {
            int victim = 0;
            faults++;
            for (j = 0; j < frames; j++) {
                if (frame[j] == -1) { victim = j; break; }
                if (last[j] < last[victim]) victim = j;
            }
            frame[victim] = pages[i];
            last[victim] = i;
        }
        printf("%d\t", pages[i]);
        for (j = 0; j < frames; j++) {
            if (frame[j] == -1) printf("- ");
            else printf("%d ", frame[j]);
        }
        printf("\t%s\n", found ? "HIT" : "FAULT");
    }
    printf("Total Page Faults = %d\n", faults);
    printf("Total Page Hits = %d\n", hits);
    printf("Hit Ratio = %.2f\n", (float)hits/n);
    printf("Fault Ratio = %.2f\n", (float)faults/n);
    return 0;
}

    for (int i = 0; i < framesCount; i++) {
        frames[i] = -1;
        lastUsed[i] = -1;
    }
    cout << "\nPage\tFrames\t\tStatus\n";
    for (int i = 0; i < n; i++) {
        int page = pages[i], found = 0;
        for (int j = 0; j < frameCount; j++) {
            if (frames[j] == page) {
                found = 1;
                lastUsed[j] = i;
                break;
            }
        }
        if (found) {
            pageHits++;
        } else {
            pageFaults++;
            if (frameCount < framesCount) {
                frames[frameCount] = page;
                lastUsed[frameCount] = i;
                frameCount++;
            } else {
                int lruIndex = 0;
                for (int j = 1; j < framesCount; j++)
                    if (lastUsed[j] < lastUsed[lruIndex])
                        lruIndex = j;
                frames[lruIndex] = page;
                lastUsed[lruIndex] = i;
            }
        }
        cout << page << "\t";
        for (int j = 0; j < frameCount; j++) cout << frames[j] << " ";
        cout << "\t" << (found ? "Hit" : "Fault") << "\n";
    }
    cout << "\nTotal Page Hits: " << pageHits;
    cout << "\nTotal Page Faults: " << pageFaults << "\n";
    return 0;
}
Compile: g++ lru.cpp -o lru && ./lru. The source manual hard-codes four frames; this program retains that setup.

Experiment 14 — Process Synchronization — Pseudocode
Your Experiment 14 manual specifically requests pseudocode with comments for three synchronization problems, not a C
implementation.
1. Producer–Consumer (bounded buffer of size 10)
shared buffer[10]
semaphore empty = 10
semaphore full = 0
semaphore mutex = 1
PRODUCER:
while true:
    item = produce_order()
    wait(empty)       // wait until a free slot exists
    wait(mutex)       // enter critical section
    insert item into buffer
    signal(mutex)     // release shared buffer
    signal(full)      // announce one available item
CONSUMER:
while true:
    wait(full)        // wait until an order exists
    wait(mutex)       // enter critical section
    item = remove item from buffer
    signal(mutex)     // release shared buffer
    signal(empty)     // announce one free slot
    prepare(item)
2. Readers–Writers (reader-preference solution)
shared int readCount = 0
semaphore mutex = 1
semaphore rw_mutex = 1
READER:
wait(mutex)
readCount = readCount + 1
if readCount == 1:
    wait(rw_mutex)       // first reader blocks writers
signal(mutex)
read_database()
wait(mutex)
readCount = readCount - 1
if readCount == 0:
    signal(rw_mutex)     // last reader allows writers
signal(mutex)
WRITER:
wait(rw_mutex)           // exclusive database access
write_database()
signal(rw_mutex)
NOTE: This reader-preference version can cause writer starvation.
3. Test-and-Set Lock (bank account critical section)
shared boolean lock = false
shared balance = 1000
function TestAndSet(lock):
    old = lock
    lock = true
    return old             // atomic operation
THREAD_TRANSACTION(amount):
    while TestAndSet(lock) == true:
        ;                  // spin / busy wait
    // Critical section: only one thread updates balance
    balance = balance + amount
    lock = false           // release lock
    // Critical section ends
Viva points: Producer–Consumer uses empty/full counting semaphores and mutex. Readers–Writers allows concurrent
readers but exclusive writers. Test-and-Set is atomic; spinning is busy waiting and may starve a thread.

CPU scheduling formulas
CT  = Completion Time
TAT = CT - AT
WT  = TAT - BT
Average TAT = sum(TAT) / number of processes
Average WT  = sum(WT) / number of processes
FCFS: execute in arrival order.
Priority: lowest priority number wins (as in your manual).
SJF: choose shortest burst among arrived processes.
SRTF: choose shortest remaining time at every time unit.
LRU: replace the page that was used least recently.
